// The Silsila Markaz's reporting website (reporting.naqshbandiaowaisiah.us).
// Plain script, no libraries: the page is built with DOM calls (never from
// names as HTML), charts are drawn as SVG, and every figure comes from the
// platform's /v1/markaz routes through this site's /api/.
(() => {
  "use strict";
  const TZ = Intl.DateTimeFormat().resolvedOptions().timeZone || "UTC";
  const SVGNS = "http://www.w3.org/2000/svg";
  const app = document.getElementById("app");
  const state = { view: "overview", preset: "30", from: "", to: "", scope: "all", kind: "all", access: null, org: null, date: "", month: "", timers: [] };

  // MARK: - Small helpers
  function h(tag, attrs, ...children) {
    const node = document.createElement(tag);
    for (const [key, value] of Object.entries(attrs || {})) {
      if (value === null || value === undefined || value === false) continue;
      if (key === "class") node.className = value;
      else if (key.startsWith("on")) node.addEventListener(key.slice(2), value);
      else node.setAttribute(key, value === true ? "" : String(value));
    }
    for (const child of children.flat(Infinity)) {
      if (child === null || child === undefined || child === false) continue;
      node.append(child instanceof Node ? child : document.createTextNode(String(child)));
    }
    return node;
  }
  function s(tag, attrs, ...children) {
    const node = document.createElementNS(SVGNS, tag);
    for (const [key, value] of Object.entries(attrs || {})) if (value !== null && value !== undefined) node.setAttribute(key, String(value));
    for (const child of children.flat(Infinity)) if (child) node.append(child instanceof Node ? child : document.createTextNode(String(child)));
    return node;
  }
  const num = value => new Intl.NumberFormat("en").format(Math.round((value || 0) * 10) / 10);
  const pct = value => value === null || value === undefined ? "–" : `${Math.round(value * 100)}%`;
  const fmt = (iso, opts) => iso ? new Intl.DateTimeFormat("en-GB", { timeZone: TZ, ...opts }).format(new Date(iso)) : "–";
  const when = iso => fmt(iso, { day: "numeric", month: "short", year: "numeric", hour: "2-digit", minute: "2-digit" });
  const day = iso => fmt(iso, { day: "numeric", month: "short", year: "numeric" });
  const time = iso => fmt(iso, { hour: "2-digit", minute: "2-digit" });
  const localDate = (d = new Date()) => new Intl.DateTimeFormat("en-CA", { timeZone: TZ, year: "numeric", month: "2-digit", day: "2-digit" }).format(d);
  const addDays = (date, n) => new Date(Date.parse(`${date}T00:00:00Z`) + n * 86400000).toISOString().slice(0, 10);
  const WEEKDAYS = ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"];
  const STATUS = { active: "Active", quiet: "Quiet", inactive: "Inactive" };

  function delta(now, before) {
    if (!before) return h("span", { class: "delta flat" }, now ? "new" : "–");
    const change = (now - before) / before;
    const cls = change > 0.005 ? "up" : change < -0.005 ? "down" : "flat";
    return h("span", { class: `delta ${cls}` }, `${change > 0 ? "▲" : change < 0 ? "▼" : "•"} ${Math.abs(Math.round(change * 100))}% vs before`);
  }
  function toast(text) {
    const node = h("div", { class: "toast", role: "status" }, text);
    document.body.append(node);
    setTimeout(() => node.remove(), 3200);
  }
  /** Replaces a node's children with the given nodes, lists flattened and empty ones left out. */
  const put = (root, ...items) => root.replaceChildren(...items.flat(Infinity).filter(item => item !== null && item !== undefined && item !== false));
  function clearTimers() { state.timers.forEach(clearInterval); state.timers = []; }

  async function api(path, options = {}) {
    const response = await fetch(`/api/${path}`, { credentials: "same-origin", ...options,
      headers: { accept: "application/json", ...(options.body ? { "content-type": "application/json" } : {}), ...(options.method && options.method !== "GET" ? { "x-markaz": "1" } : {}) } });
    if (response.status === 401) { signIn(); throw new Error("signed-out"); }
    const body = response.status === 204 ? {} : await response.json().catch(() => ({}));
    if (!response.ok) throw new Error(body.error || `Request failed (${response.status})`);
    return body;
  }

  function range() {
    const today = localDate();
    if (state.preset === "custom" && state.from && state.to) return { from: state.from, to: state.to };
    if (state.preset === "month") return { from: `${today.slice(0, 7)}-01`, to: today };
    if (state.preset === "lastmonth") {
      const end = addDays(`${today.slice(0, 7)}-01`, -1);
      return { from: `${end.slice(0, 7)}-01`, to: end };
    }
    const days = Number(state.preset) || 30;
    return { from: addDays(today, -(days - 1)), to: today };
  }
  const query = extra => {
    const r = range();
    const params = new URLSearchParams({ from: r.from, to: r.to, tz: TZ, scope: state.scope, kind: state.kind, ...(extra || {}) });
    return params.toString();
  };

  // MARK: - Sign in
  let polling = null;
  async function signIn() {
    clearTimers();
    if (polling) clearTimeout(polling);
    const code = h("div", { class: "code", "aria-live": "polite" }, "· · · ·");
    const wait = h("div", { class: "wait" }, h("span", { class: "dot" }), h("span", {}, "Waiting for approval in the app…"));
    const left = h("span", {});
    app.replaceChildren(h("div", { class: "signin" },
      h("div", { class: "side" },
        h("div", { class: "brand" }, h("img", { src: "/logo.webp", alt: "" }), h("div", {}, h("b", {}, "Silsila Naqshbandia Owaisiah"), h("div", {}, "Silsila Markaz"))),
        h("div", {}, h("h1", {}, "Every Halqah, in one place."),
          h("div", { class: "points" },
            h("div", {}, h("b", {}, "◆"), h("span", {}, "Daily and monthly Zikr reports: who attended, where and for how long.")),
            h("div", {}, h("b", {}, "◆"), h("span", {}, "Trends, the busiest hours, the regulars, and those not seen for a while.")),
            h("div", {}, h("b", {}, "◆"), h("span", {}, "The whole organisation by region, live sessions now, and notices to many Halqahs at once.")))),
        h("div", { class: "updated" }, "For super admins and the officers of the Silsila Markaz.")),
      h("div", { class: "card" },
        h("h2", {}, "Sign in"),
        h("p", { class: "sub" }, "Approve this computer from the Darul Irfan app."),
        code,
        h("ol", { class: "steps" },
          h("li", {}, "Open Darul Irfan on your phone and go to the Halqah tab."),
          h("li", {}, "Open Reports & analytics, then Sign in on a computer."),
          h("li", {}, "Type the code above and tap Approve.")),
        wait, h("div", { class: "updated", style: "margin-top:8px" }, left))));
    try {
      const response = await fetch("/auth/start", { method: "POST" });
      const link = await response.json();
      if (!response.ok || !link.code) throw new Error(link.error || "Could not start");
      code.textContent = link.code;
      const expires = Date.parse(link.expiresAt);
      const tick = async () => {
        const seconds = Math.max(0, Math.round((expires - Date.now()) / 1000));
        left.textContent = `This code works for ${Math.floor(seconds / 60)}:${String(seconds % 60).padStart(2, "0")} more.`;
        if (seconds <= 0) return signIn();
        const poll = await fetch("/auth/poll", { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ pollToken: link.pollToken }) })
          .then(r => r.json()).catch(() => ({ status: "pending" }));
        if (poll.status === "approved") { toast("Signed in"); return boot(); }
        if (poll.status === "expired") return signIn();
        polling = setTimeout(tick, 2500);
      };
      polling = setTimeout(tick, 1500);
    } catch (error) {
      wait.replaceChildren(h("span", { class: "err" }, "The sign-in service is unavailable. Refresh in a moment."));
    }
  }

  // MARK: - Shell
  const VIEWS = [
    ["overview", "Overview", "◈"], ["daily", "Daily report", "☀"], ["monthly", "Monthly report", "▦"],
    ["halqahs", "Halqahs", "◎"], ["people", "People", "☺"], ["sessions", "Sessions", "◷"],
    ["organisation", "Organisation", "⌂"], ["notices", "Notices", "✉"], ["computers", "Signed-in computers", "⎙"],
  ];

  async function boot() {
    try {
      state.access = await api("access");
      if (!state.access.canReport) {
        app.replaceChildren(h("div", { class: "signin" }, h("div", { class: "card" }, h("h2", {}, "No access"),
          h("p", {}, "This account is not a super admin or an officer of the Silsila Markaz."),
          h("button", { class: "btn em", onclick: signOut }, "Sign out"))));
        return;
      }
      state.org = await api("organization");
      const view = location.hash.slice(1);
      if (VIEWS.some(([id]) => id === view)) state.view = view;
      render();
    } catch (error) {
      if (error.message !== "signed-out") app.replaceChildren(h("div", { class: "boot" }, h("p", { class: "err" }, error.message)));
    }
  }

  async function signOut() {
    await fetch("/auth/signout", { method: "POST" }).catch(() => null);
    signIn();
  }

  function scopeSelect() {
    const select = h("select", { "aria-label": "Halqahs", onchange: event => { state.scope = event.target.value; render(); } },
      h("option", { value: "all" }, "All Halqahs"));
    for (const region of state.org?.regions || []) {
      const group = h("optgroup", { label: region.name }, h("option", { value: `region:${region.name}` }, `All of ${region.name}`));
      for (const halqah of region.halqahs) group.append(h("option", { value: `halqah:${halqah.id}` }, halqah.name));
      select.append(group);
    }
    const loose = state.org?.unassigned || [];
    if (loose.length) select.append(h("optgroup", { label: "Not placed in a region" }, loose.map(item => h("option", { value: `halqah:${item.id}` }, item.name))));
    select.append(h("option", { value: "halqah:silsila-markaz" }, "Silsila Markaz"));
    select.value = state.scope;
    return select;
  }

  function filters() {
    const presets = [["7", "7 days"], ["30", "30 days"], ["90", "90 days"], ["365", "12 months"], ["month", "This month"], ["lastmonth", "Last month"], ["custom", "Custom"]];
    const preset = h("select", { "aria-label": "Period", onchange: event => { state.preset = event.target.value; if (state.preset !== "custom") render(); else render(); } },
      presets.map(([value, label]) => h("option", { value }, label)));
    preset.value = state.preset;
    const r = range();
    const custom = state.preset === "custom" ? [
      h("input", { type: "date", value: state.from || r.from, "aria-label": "From", onchange: event => { state.from = event.target.value; render(); } }),
      h("input", { type: "date", value: state.to || r.to, "aria-label": "To", onchange: event => { state.to = event.target.value; render(); } }),
    ] : [];
    const kind = h("select", { "aria-label": "Kind", onchange: event => { state.kind = event.target.value; render(); } },
      h("option", { value: "all" }, "Zikr and meetings"), h("option", { value: "zikr" }, "Zikr only"), h("option", { value: "meeting" }, "Meetings only"));
    kind.value = state.kind;
    return h("div", { class: "filters" }, preset, custom, scopeSelect(), kind);
  }

  function render() {
    clearTimers();
    location.hash = state.view;
    const nav = h("nav", { class: "nav", "aria-label": "Sections" },
      h("div", { class: "brand" }, h("img", { src: "/logo.webp", alt: "" }), h("div", {}, h("b", {}, "Silsila Markaz"), h("span", {}, "Reports & analytics"))),
      VIEWS.slice(0, 6).map(([id, label, icon]) => h("a", { href: `#${id}`, class: state.view === id ? "on" : "", onclick: event => { event.preventDefault(); go(id); } }, h("span", {}, icon), label)),
      h("div", { class: "sep" }, "Organisation"),
      VIEWS.slice(6).map(([id, label, icon]) => h("a", { href: `#${id}`, class: state.view === id ? "on" : "", onclick: event => { event.preventDefault(); go(id); } }, h("span", {}, icon), label)),
      h("div", { class: "who" }, state.access?.isSuperAdmin ? "Signed in as a super admin" : "Signed in as a Markaz officer",
        h("div", { style: "margin-top:8px;display:flex;gap:8px;flex-wrap:wrap" },
          h("button", { class: "btn", onclick: toggleTheme }, "Theme"), h("button", { class: "btn", onclick: signOut }, "Sign out"))));
    const title = VIEWS.find(([id]) => id === state.view)?.[1] || "Overview";
    const content = h("div", { class: "content" }, h("div", { class: "card" }, h("p", { class: "empty" }, "Loading…")));
    const usesFilters = ["overview", "halqahs", "people", "sessions"].includes(state.view);
    const top = h("header", { class: "top" },
      h("button", { class: "menu", onclick: () => nav.classList.toggle("open"), "aria-label": "Menu" }, "☰"),
      h("h1", {}, title), usesFilters ? filters() : null);
    app.replaceChildren(h("div", { class: "shell" }, nav, h("main", { class: "main" }, top, content)));
    const views = { overview, daily: dailyView, monthly: monthlyView, halqahs: halqahsView, people: peopleView, sessions: sessionsView,
      organisation: organisationView, notices: noticesView, computers: computersView };
    (views[state.view] || overview)(content).catch(error => {
      if (error.message !== "signed-out") content.replaceChildren(h("div", { class: "card" }, h("p", { class: "err" }, error.message)));
    });
  }
  function go(view) { state.view = view; render(); }
  function toggleTheme() {
    const root = document.documentElement;
    const dark = root.dataset.theme ? root.dataset.theme === "dark" : matchMedia("(prefers-color-scheme: dark)").matches;
    root.dataset.theme = dark ? "light" : "dark";
    try { localStorage.setItem("markaz-theme", root.dataset.theme); } catch (_) { /* private mode */ }
  }

  // MARK: - Charts
  function trendChart(days) {
    const width = 900, height = 260, pad = { l: 40, r: 40, t: 14, b: 28 };
    const maxA = Math.max(1, ...days.map(d => d.attendance)), maxS = Math.max(1, ...days.map(d => d.sessions));
    const x = i => pad.l + (days.length <= 1 ? 0 : (i / (days.length - 1)) * (width - pad.l - pad.r));
    const y = v => height - pad.b - (v / maxA) * (height - pad.t - pad.b);
    const svg = s("svg", { viewBox: `0 0 ${width} ${height}`, class: "chart", role: "img",
      "aria-label": `Attendance per day, from ${days[0]?.date} to ${days[days.length - 1]?.date}; highest ${maxA}.` });
    const defs = s("defs", {}, s("linearGradient", { id: "area", x1: 0, y1: 0, x2: 0, y2: 1 },
      s("stop", { offset: "0%", "stop-color": "#0f7a59", "stop-opacity": ".45" }), s("stop", { offset: "100%", "stop-color": "#0f7a59", "stop-opacity": "0" })));
    svg.append(defs);
    for (let i = 0; i <= 4; i++) {
      const v = Math.round((maxA * i) / 4), yy = y(v);
      svg.append(s("line", { x1: pad.l, x2: width - pad.r, y1: yy, y2: yy, stroke: "currentColor", "stroke-opacity": ".08" }), s("text", { x: pad.l - 6, y: yy + 4, "text-anchor": "end" }, v));
    }
    const barW = Math.max(2, (width - pad.l - pad.r) / Math.max(1, days.length) * 0.5);
    days.forEach((d, i) => {
      if (!d.sessions) return;
      const bh = (d.sessions / maxS) * (height - pad.t - pad.b) * 0.35;
      svg.append(s("rect", { x: x(i) - barW / 2, y: height - pad.b - bh, width: barW, height: bh, rx: 2, fill: "#c79a3a", "fill-opacity": ".55" },
        s("title", {}, `${d.date}: ${d.sessions} sessions`)));
    });
    if (days.length) {
      const line = days.map((d, i) => `${i ? "L" : "M"}${x(i).toFixed(1)},${y(d.attendance).toFixed(1)}`).join(" ");
      svg.append(s("path", { d: `${line} L${x(days.length - 1)},${height - pad.b} L${x(0)},${height - pad.b} Z`, fill: "url(#area)" }));
      svg.append(s("path", { d: line, fill: "none", stroke: "#0f7a59", "stroke-width": 2.5, "stroke-linejoin": "round" }));
      days.forEach((d, i) => svg.append(s("circle", { cx: x(i), cy: y(d.attendance), r: days.length > 60 ? 1.5 : 3, fill: "#0b5c43" },
        s("title", {}, `${d.date}: ${d.attendance} present, ${d.unique} different people, ${d.sessions} sessions`))));
      const step = Math.max(1, Math.ceil(days.length / 8));
      days.forEach((d, i) => { if (i % step === 0 || (i === days.length - 1 && i % step >= step / 2)) svg.append(s("text", { x: x(i), y: height - 8, "text-anchor": "middle" }, d.date.slice(5))); });
    }
    return h("div", {}, svg, h("div", { class: "legend" }, h("span", {}, h("i", { style: "background:#0f7a59" }), "Attendance"), h("span", {}, h("i", { style: "background:#c79a3a" }), "Sessions")));
  }

  function heatmap(cells) {
    if (!cells.length) return h("p", { class: "empty" }, "No sessions in this period.");
    const hours = [...new Set(cells.map(c => c.hour))].sort((a, b) => a - b);
    const max = Math.max(1, ...cells.map(c => c.attendance));
    const grid = h("div", { class: "heat", style: `grid-template-columns:44px repeat(${hours.length},minmax(26px,1fr))`, role: "table", "aria-label": "Attendance by weekday and hour" });
    grid.append(h("div", {}), ...hours.map(hour => h("div", { style: "text-align:center" }, `${String(hour).padStart(2, "0")}:00`)));
    for (let w = 1; w <= 7; w++) {
      grid.append(h("div", {}, WEEKDAYS[w - 1]));
      for (const hour of hours) {
        const cell = cells.find(c => c.weekday === w && c.hour === hour);
        const value = cell ? cell.attendance : 0;
        const alpha = value ? 0.18 + 0.82 * (value / max) : 0.05;
        grid.append(h("div", { class: "cell", style: `background:rgba(11,92,67,${alpha.toFixed(2)})`, title: cell ? `${WEEKDAYS[w - 1]} ${hour}:00 · ${cell.sessions} sessions · ${value} present` : "" }, value || ""));
      }
    }
    return h("div", { class: "scroll" }, grid);
  }

  function bars(rows, value, label, onpick) {
    const max = Math.max(1, ...rows.map(value));
    if (!rows.length) return h("p", { class: "empty" }, "Nothing to show.");
    return h("div", { class: "bars" }, rows.map(row => h("div", { class: "row", onclick: onpick ? () => onpick(row) : null, title: onpick ? "Show only this" : "" },
      h("span", {}, label(row)), h("div", { class: "track" }, h("div", { class: "fill", style: `width:${(value(row) / max) * 100}%` })), h("span", { class: "num" }, num(value(row))))));
  }

  function donut(app, web) {
    const total = Math.max(1, app + web), r = 70, c = 2 * Math.PI * r, share = app / total;
    return h("div", { style: "display:flex;gap:22px;align-items:center;flex-wrap:wrap" },
      s("svg", { viewBox: "0 0 180 180", width: 170, height: 170, role: "img", "aria-label": `${app} through the app and ${web} from the websites` },
        s("circle", { cx: 90, cy: 90, r, fill: "none", stroke: "#c79a3a", "stroke-width": 24 }),
        s("circle", { cx: 90, cy: 90, r, fill: "none", stroke: "#0b5c43", "stroke-width": 24, "stroke-dasharray": `${c * share} ${c}`, transform: "rotate(-90 90 90)" }),
        s("text", { x: 90, y: 88, "text-anchor": "middle", style: "font:700 26px Georgia;fill:currentColor" }, Math.round(share * 100) + "%"),
        s("text", { x: 90, y: 108, "text-anchor": "middle", style: "font-size:11px;fill:currentColor;opacity:.7" }, "in the app")),
      h("div", { class: "legend", style: "display:grid" }, h("span", {}, h("i", { style: "background:#0b5c43" }), `App · ${num(app)}`), h("span", {}, h("i", { style: "background:#c79a3a" }), `Websites · ${num(web)}`)));
  }

    const card = (label, value, before, note) => h("div", { class: "card kpi" }, h("div", { class: "label" }, label), h("div", { class: "value" }, value),

      card("Different people", summary.uniqueAttendees, previous?.uniqueAttendees),
      card("Average per session", num(summary.averageAttendance), previous?.averageAttendance),
      card("From the websites", summary.webAttendance, undefined, "Visitors not in the app"),
      card("Average length", `${summary.averageMinutes} min`, undefined, `${num(summary.totalMinutes / 60)} hours in all`),
      card("Active Halqahs", `${summary.activeHalqahs} / ${summary.totalHalqahs}`, undefined, "held at least one session"),

  }

  // MARK: - Views
  async function overview(root) {

  // MARK: - Modals
  function modal(title, ...body) {
    const close = () => { node.remove(); document.removeEventListener("keydown", onkey); };
    const onkey = event => { if (event.key === "Escape") close(); };
    const node = h("div", { class: "modal", role: "dialog", "aria-modal": "true", "aria-label": title, onclick: event => { if (event.target === node) close(); } },
      h("div", { class: "box" }, h("div", { class: "head" }, h("h2", {}, title), h("button", { class: "btn", onclick: close, "aria-label": "Close" }, "✕")), ...body));
    document.addEventListener("keydown", onkey);
    document.body.append(node);
    return close;
  }

      h("div", { class: "row" }, h("button", { class: "btn em", onclick: async () => {
        try {
          await api(`halqahs/${item.id}`, { method: "PUT", body: JSON.stringify({ region: region.value.trim() || null, countryCode: country.value.trim() || null, city: city.value.trim() || null }) });
          close(); toast("Saved"); render();
        } catch (error) { toast(error.message); }
      } }, "Save"))));
  }

  function announceModal() {
    const org = state.org || { regions: [], unassigned: [] };
    const all = [...org.regions.flatMap(r => r.halqahs), ...org.unassigned];
    const title = h("input", { type: "text", maxlength: 200, placeholder: "Title" });
    const body = h("textarea", { maxlength: 4000, placeholder: "The notice" });
    const audience = h("select", {}, h("option", { value: "members" }, "Members of the chosen Halqahs"), h("option", { value: "admins" }, "Every Halqah's admins (through the Markaz)"));
    const scope = h("select", {}, h("option", { value: "all" }, "Every Halqah"), org.regions.map(r => h("option", { value: `region:${r.name}` }, `All of ${r.name}`)), h("option", { value: "halqahs" }, "Choose Halqahs"));
    const picks = h("div", { style: "display:none;grid-template-columns:repeat(auto-fill,minmax(200px,1fr));gap:6px" },
      all.map(item => h("label", { style: "display:flex;gap:6px;font-weight:500" }, h("input", { type: "checkbox", value: item.id }), item.name)));
    scope.addEventListener("change", () => { picks.style.display = scope.value === "halqahs" ? "grid" : "none"; });
    const notify = h("input", { type: "checkbox", checked: true });
    const close = modal("Notice to Halqahs", h("div", { class: "form" },
      h("label", {}, "Title", title), h("label", {}, "Message", body),
      h("div", { class: "row" }, h("label", {}, "Who receives it", audience), h("label", {}, "Halqahs", scope)), picks,
      h("label", { style: "display:flex;gap:8px;align-items:center;font-weight:500" }, notify, "Also send it as a notification to their phones"),
      h("div", { class: "row" }, h("button", { class: "btn gold", onclick: async event => {
        if (!title.value.trim()) return toast("Give the notice a title.");
        event.target.disabled = true;
        try {
          const chosen = [...picks.querySelectorAll("input:checked")].map(input => input.value);
          const result = await api("announce", { method: "POST", body: JSON.stringify({ title: title.value, body: body.value, audience: audience.value,
            scope: scope.value, halqahIDs: chosen, notify: notify.checked }) });
          close(); toast(`Sent to ${result.posted} ${result.posted === 1 ? "group" : "Halqahs"} · ${result.notified} phones notified`);
          if (state.view === "notices") render();
        } catch (error) { toast(error.message); event.target.disabled = false; }
      } }, "Send"))));
  }

  // MARK: - Start
  try { const theme = localStorage.getItem("markaz-theme"); if (theme) document.documentElement.dataset.theme = theme; } catch (_) { /* private mode */ }
  addEventListener("hashchange", () => {
    const view = location.hash.slice(1);
    if (state.access && view !== state.view && VIEWS.some(([id]) => id === view)) go(view);
  });
  boot();
